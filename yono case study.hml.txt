  function logEvent(event, meta = {}) {
    const time = new Date().toLocaleTimeString([], { hour12: false });
    state.telemetry.unshift({ time, event, meta });
    renderTelemetry();
  }

  function renderTelemetry() {
    const box = document.getElementById('telemetryBox');
    box.innerHTML = state.telemetry.map(t => `
      <div class="tel-entry">
        <span style="color:#9ca3af">[${t.time}]</span>
        <span class="tel-event">${t.event}</span>
        <div class="tel-safe">${JSON.stringify(t.meta)}</div>
      </div>
    `).join('');
  }

  function setDashMode(mode) {
    state.dashMode = mode;
    document.querySelectorAll('[id^="mode-"]').forEach(el => el.classList.remove('active'));
    document.getElementById('mode-' + mode).classList.add('active');

    const net = document.getElementById('netBadge');
    net.textContent = mode === 'degraded' ? 'High Load • Critical Only' :
                      mode === 'stale' ? 'Offline / Cached' : '4G • Secure';

    if (mode === 'slow_secondary') {
      state.secondaryLoaded = false;
      render();
      setTimeout(() => {
        if (state.dashMode === 'slow_secondary') {
          state.secondaryLoaded = true;
          render();
        }
      }, 2800);
    } else {
      state.secondaryLoaded = true;
      render();
    }
  }

  function setOutcomeMode(mode) {
    state.outcomeMode = mode;
    document.querySelectorAll('[id^="out-"]').forEach(el => el.classList.remove('active'));
    document.getElementById('out-' + mode).classList.add('active');
  }

  function simulateInterruptedSession() {
    if (!state.loggedIn) return;
    state.loggedIn = false;
    state.sessionExpiredNotice = true;
    logEvent('task_abandoned', { step: state.screen, reason: 'session_expiry' });
    navigate('login');
  }

  function resetPrototype() {
    state.loggedIn = false;
    state.pendingTx = null;
    state.lastTxResult = null;
    state.sessionExpiredNotice = false;
    state.searchQuery = '';
    setDashMode('normal');
    setOutcomeMode('success');
    state.telemetry = [];
    logEvent('app_launch', { app_version: '2.0.0-proto', os_cohort: 'web_sim', network_class: '4g' });
    navigate('login');
  }

  function navigate(screen) {
    state.screen = screen;
    render();
  }

  function handleLogin() {
    logEvent('login_started', { method: 'mpin_demo' });
    state.loggedIn = true;
    state.sessionExpiredNotice = false;
    logEvent('login_success', { duration_ms: 420, credentials_logged: false });

    // FR-03 / NFR: Essential dashboard usable independently
    const dashStart = performance.now();
    if (state.dashMode === 'slow_secondary') state.secondaryLoaded = false;
    navigate('dashboard');
    logEvent('dashboard_usable', {
      time_to_usable_ms: Math.round(performance.now() - dashStart + 310),
      critical_ready: true,
      secondary_ready: state.secondaryLoaded
    });

    if (state.dashMode === 'slow_secondary') {
      setTimeout(() => {
        state.secondaryLoaded = true;
        if (state.screen === 'dashboard') render();
      }, 2500);
    }
  }

  function refreshAccountData() {
    if (state.dashMode === 'stale') {
      setDashMode('normal');
      showToast('Account balance refreshed from core banking.');
    } else {
      showToast('Account data is already up to date.');
    }
  }

  function showToast(msg) {
    const existing = document.getElementById('toastBanner');
    if (existing) existing.remove();
    const div = document.createElement('div');
    div.id = 'toastBanner';
    div.className = 'banner banner-info';
    div.textContent = msg;
    document.getElementById('appBody').prepend(div);
    setTimeout(() => div.remove(), 3500);
  }

  function selectBene(id) {
    state.selectedBeneId = id;
    render();
  }

  function goToReview() {
    const amt = Number(document.getElementById('amtInput').value);
    if (!amt || amt <= 0 || amt > 100000) {
      showToast('Please enter a valid amount between ₹1 and ₹1,00,000.');
      return;
    }
    const bene = state.beneficiaries.find(b => b.id === state.selectedBeneId);
    if (!bene || !bene.available) {
      showToast('Selected beneficiary is currently unavailable. Please choose another payee.');
      return;
    }
    state.amount = amt;
    state.rail = document.getElementById('railSelect').value;
    logEvent('transaction_started', { journey: 'fund_transfer', rail: state.rail });
    navigate('review');
  }

  function submitTransfer(simulateCrash = false) {
    // FR-06: Client-side interaction debounce + idempotency key
    if (state.isSubmitting) return;
    state.isSubmitting = true;

    const idempotencyKey = 'idem_' + Math.random().toString(36).slice(2, 10);
    const bene = state.beneficiaries.find(b => b.id === state.selectedBeneId);

    logEvent('transaction_submitted', {
      journey: 'fund_transfer',
      rail: state.rail,
      correlation_id: idempotencyKey
    });

    render(); // Shows processing state & disabled button

    setTimeout(() => {
      state.isSubmitting = false;

      // FR-07 Edge Case: App closed / interrupted mid-transfer
      if (simulateCrash) {
        state.pendingTx = {
          refId: 'SBI' + Math.floor(100000000 + Math.random() * 900000000),
          idempotencyKey,
          to: bene.name,
          amount: Number(state.amount),
          rail: state.rail,
          timestamp: 'Just now',
          status: 'pending',
          source: 'Awaiting Core Banking / NPCI Reconciliation',
          reason: 'Connection was interrupted before confirmation arrived.'
        };
        state.transactions.unshift(state.pendingTx);
        logEvent('task_abandoned', { step: 'authorise_submit', correlation_id: idempotencyKey });
        alert('Simulated App Close! Reopening YONO SBI to test FR-07 Safe Recovery.');
        navigate('dashboard');
        return;
      }

      const refId = 'SBI' + Math.floor(100000000 + Math.random() * 900000000);
      const outcome = state.outcomeMode;

      const tx = {
        refId,
        idempotencyKey,
        to: bene.name,
        amount: Number(state.amount),
        rail: state.rail,
        status: outcome,
        timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        source: outcome === 'pending' ? 'Awaiting Authoritative Switch' : 'SBI Core Banking / NPCI',
        reason: outcome === 'success'
          ? 'Amount has been credited to the beneficiary account.'
          : outcome === 'pending'
          ? 'We did not receive final confirmation within the timeout window. Do not submit again until status is checked.'
          : 'Beneficiary bank declined the request (Limit/Account rule). No money was debited.'
      };

      state.transactions.unshift(tx);
      state.lastTxResult = tx;
      if (outcome === 'pending') {
        state.pendingTx = tx;
      }

      logEvent(`transaction_${outcome}`, {
        rail: state.rail,
        authoritative: outcome !== 'pending',
        correlation_id: idempotencyKey
      });

      navigate('outcome');
    }, 900);
  }

  function reconcilePendingTx(refId) {
    const tx = state.transactions.find(t => t.refId === refId);
    if (!tx) return;

    logEvent('retry_attempted', { action: 'reconcile_first', original_state: tx.status, refId });

    // Reconcile resolves pending -> authoritative success (or failed)
    tx.status = 'success';
    tx.source = 'NPCI Reconciled Authoritative Status';
    tx.reason = 'Reconciliation complete: Original transfer succeeded. No second transfer is needed.';
    if (state.pendingTx && state.pendingTx.refId === refId) {
      state.pendingTx = null;
    }
    state.lastTxResult = tx;
    logEvent('transaction_success', { reconciled: true, refId });
    navigate('outcome');
  }
